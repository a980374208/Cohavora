cmake_minimum_required(VERSION 3.25)
project(ProductGpuBudgetWitness LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The DXGI budget witness requires Windows")
endif()
add_executable(product_gpu_budget_probe ../../../probes/product_gpu_budget_probe.cpp)
target_compile_features(product_gpu_budget_probe PRIVATE cxx_std_20)
target_link_libraries(product_gpu_budget_probe PRIVATE dxgi)

add_executable(product_gpu_trace ../../../probes/product_gpu_trace.cpp)
target_compile_features(product_gpu_trace PRIVATE cxx_std_20)
target_link_libraries(product_gpu_trace PRIVATE advapi32 tdh)

add_executable(test_product_gpu_trace_publication ../../../selftests/test_product_gpu_trace_publication.cpp)
target_compile_features(test_product_gpu_trace_publication PRIVATE cxx_std_20)
target_link_libraries(test_product_gpu_trace_publication PRIVATE advapi32 tdh)
enable_testing()
add_test(NAME product_gpu_trace_publication COMMAND test_product_gpu_trace_publication)
set_tests_properties(product_gpu_trace_publication PROPERTIES TIMEOUT 10)
