cmake_minimum_required(VERSION 3.25)
project(ProductMicrophoneInputWitness LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This independent microphone witness requires Windows")
endif()
add_executable(product_microphone_input ../../../probes/product_microphone_input.cpp)
target_compile_features(product_microphone_input PRIVATE cxx_std_20)
target_compile_definitions(product_microphone_input PRIVATE UNICODE _UNICODE WITNESS_CONFIGURATION="$<CONFIG>")
target_link_libraries(product_microphone_input PRIVATE ole32 uuid bcrypt)
add_executable(microphone_signal_metrics_test ../../../selftests/test_microphone_signal_metrics.cpp)
target_compile_features(microphone_signal_metrics_test PRIVATE cxx_std_20)
enable_testing()
add_test(NAME microphone_signal_metrics COMMAND microphone_signal_metrics_test)
